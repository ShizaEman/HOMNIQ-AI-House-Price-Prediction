@st.cache_resource
def load_model():
    if os.path.exists(MODEL_PATH):
        try:
            with open(MODEL_PATH, "rb") as f:
                return pickle.load(f)
        except Exception:
            return None
    return None


MODEL = load_model()


def predict_price(values: dict):
    """values keys must match FEATURE_ORDER"""
    ordered = [values[f] for f in FEATURE_ORDER]
    if MODEL is not None:
        pred = MODEL.predict(np.array([ordered]))[0]
        return float(pred)
    # Fallback demo estimate if no model file is found, so the UI still works
    base = 20000
    demo = (
        base
        + values["OverallQual"] * 18000
        + values["GrLivArea"] * 55
        + values["GarageCars"] * 9000
        + values["TotalBsmtSF"] * 20
        + (values["YearBuilt"] - 1900) * 150
        + values["FullBath"] * 6000
        + values["BedroomAbvGr"] * 4000
        + values["LotArea"] * 1.2
    )
    return float(demo)

def render_navbar():

            "<span style='font-size:11px;color:#8a93a6;'>Smart Valuation for Your Dream Home</span></div>"
            "</div>",
            unsafe_allow_html=True,
        )

    pages = ["Home", "Prediction", "Insights", "About"]
    cols = [nav1, nav2, nav3, nav4]
    for col, p in zip(cols, pages):
        with col:
            st.markdown("<div style='padding-top:14px;'>", unsafe_allow_html=True)
            if st.session_state.page == p or (p == "Prediction" and st.session_state.page == "Result"):

                if st.button(p, key=f"nav_{p}"):

            st.markdown("</div>", unsafe_allow_html=True)

    with cta_col:
        st.markdown("<div style='padding-top:8px;'>", unsafe_allow_html=True)
        if st.button("Get Started", key="nav_cta", type="primary"):
            go_to("Prediction")
        st.markdown("</div>", unsafe_allow_html=True)

    st.markdown("<hr>", unsafe_allow_html=True)

def prediction_page():
    st.markdown("<div class='page-title'>🏠 House Price Prediction</div>", unsafe_allow_html=True)
    st.markdown("<div class='page-sub'>Enter property details to get an instant price estimate</div>", unsafe_allow_html=True)

    if MODEL is None:
        st.warning(
            f"Model file '{MODEL_PATH}' not found in the app directory. "
            "Running in demo mode with an estimated formula — place your trained "
            "`house_price_model.pkl` next to app.py to enable real predictions."
        )

    form_col, side_col = st.columns([1.5, 1], gap="large")

    with form_col:
        st.markdown("<div class='card'>", unsafe_allow_html=True)
        st.markdown("<div class='section-label'>📋 Property Details</div>", unsafe_allow_html=True)

        c1, c2 = st.columns(2)
        with c1:
            overall_qual = st.selectbox("Overall Quality", list(range(1, 11)), index=5)
            living_area = st.number_input("Living Area (sq ft)", min_value=100, max_value=10000, value=1800, step=50)
            garage_cap = st.selectbox("Garage Capacity", list(range(0, 6)), index=2)
            year_built = st.number_input("Year Built", min_value=1870, max_value=2026, value=2005, step=1)

        with c2:
            basement_area = st.number_input("Basement Area (sq ft)", min_value=0, max_value=6000, value=900, step=50)
            full_bath = st.selectbox("Full Bathrooms", list(range(0, 6)), index=2)
            bedrooms = st.selectbox("Bedrooms", list(range(0, 10)), index=3)
            lot_area = st.number_input("Lot Area (sq ft)", min_value=500, max_value=100000, value=8500, step=100)

        st.write("")
        predict_clicked = st.button("Predict House Price →", type="primary", use_container_width=True)
        st.markdown("</div>", unsafe_allow_html=True)

        if predict_clicked:
            inputs = {
                "OverallQual": overall_qual,
                "GrLivArea": living_area,
                "GarageCars": garage_cap,
                "TotalBsmtSF": basement_area,
                "YearBuilt": year_built,
                "FullBath": full_bath,
                "BedroomAbvGr": bedrooms,
                "LotArea": lot_area,
            }
            st.session_state.inputs = inputs
            st.session_state.prediction = predict_price(inputs)
            go_to("Result")
            st.rerun()

    with side_col:
        st.image(FORM_IMG, use_container_width=True)
        st.markdown("<div class='card' style='margin-top:16px;'>", unsafe_allow_html=True)
        st.markdown("<div style='font-weight:700;color:#0f172a;font-size:16px;'>Why Choose Our Model?</div>", unsafe_allow_html=True)
        for point in ["Trained on real estate data", "High prediction accuracy", "Easy to use interface", "Instant results"]:
            st.markdown(f"<div style='color:#4b5768;font-size:14px;margin-top:8px;'>✅ {point}</div>", unsafe_allow_html=True)
        st.markdown("</div>", unsafe_allow_html=True)

def insights_page():
    st.markdown("<div class='page-title'>📈 Model Performance Insights</div>", unsafe_allow_html=True)
    st.markdown("<div class='page-sub'>Comparison of different Machine Learning models for house price prediction</div>", unsafe_allow_html=True)

    chart_col, table_col = st.columns([1.3, 1], gap="large")

    with chart_col:
        st.markdown("<div class='card'>", unsafe_allow_html=True)
        st.markdown("<div class='section-label'>Model Comparison (R² Score)</div>", unsafe_allow_html=True)
        fig = go.Figure(
            data=[
                go.Bar(
                    x=MODEL_SCORES["Model"],
                    y=MODEL_SCORES["R2"],
                    text=[f"{v:.2f}" for v in MODEL_SCORES["R2"]],
                    textposition="outside",
                    marker_color=["#1d4ed8", "#3b82f6", "#60a5fa", "#93c5fd"],
                )
            ]
        )
        fig.update_layout(
            yaxis_range=[0, 1.05],
            plot_bgcolor="white",
            paper_bgcolor="white",
            margin=dict(l=10, r=10, t=10, b=10),
            height=360,
            font=dict(family="Poppins", color="#334155"),
        )
        st.plotly_chart(fig, use_container_width=True)
        st.markdown("</div>", unsafe_allow_html=True)

    with table_col:
        st.markdown("<div class='card'>", unsafe_allow_html=True)
        st.markdown("<div class='section-label'>Model Performance</div>", unsafe_allow_html=True)
        display_df = MODEL_SCORES.copy()
        display_df["MAE"] = display_df["MAE"].apply(lambda v: f"${v:,.0f}")
        display_df["RMSE"] = display_df["RMSE"].apply(lambda v: f"${v:,.0f}")
        display_df = display_df.rename(columns={"R2": "R² Score"})
        display_df["R² Score"] = MODEL_SCORES["R2"].apply(lambda v: f"{v:.4f}")
        st.dataframe(display_df, hide_index=True, use_container_width=True)
        st.markdown("</div>", unsafe_allow_html=True)

    st.write("")
    st.info("ℹ️ Linear Regression achieved the best performance on the test set and is selected as the final model for this project.")
